    \end{successbox}
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \item \textbf{Tiền lọc Metadata (Pre-filtering):} Sử dụng cơ sở dữ liệu quan hệ (SQL) để thu hẹp không gian tìm kiếm, loại bỏ sớm các tài liệu không liên quan trước khi tính toán độ tương đồng.
        \item \textbf{Phân bổ trọng số thời gian (Time Decay):} Áp dụng hàm suy giảm để giảm trọng số các tài liệu cũ, đảm bảo tính cập nhật của thông tin truy xuất.
        \item \textbf{Truy xuất lai (Hybrid Retrieval):} Kết hợp kỹ thuật Sparse (BM25) và Dense (Embedding) nhằm tối ưu khả năng đối khớp từ vựng và ngữ nghĩa.
    \end{itemize}
\end{frame}

\begin{frame}{Phân hệ 1: Phân tách Lưu trữ (Storage Isolation)}
    \begin{columns}[T]
        \begin{column}{.48\textwidth}
            \begin{infobox}[\faDatabase \hspace{2pt} Relational DB (SQLite)]
                \footnotesize
                \begin{itemize}
                    \setlength\itemsep{0.3em}
                    \item \textbf{Lưu trữ Metadata:} Ngày tạo, Tác giả, Trạng thái tài liệu.
                    \item \textbf{Cập nhật hiệu quả:} Cho phép cập nhật trạng thái văn bản nhanh chóng bằng truy vấn SQL mà không cần tính toán lại Vector.
                    \item \textbf{Chỉ mục văn bản:} Tích hợp FTS5 hỗ trợ thuật toán BM25.
                \end{itemize}
            \end{infobox}
        \end{column}%
        \hfill%
        \begin{column}{.48\textwidth}
            \begin{infobox}[\faVectorSquare \hspace{2pt} Vector DB (LanceDB)]
                \footnotesize
                \begin{itemize}
                    \setlength\itemsep{0.3em}
                    \item \textbf{Lưu trữ Embeddings:} Chuyên biệt cho biểu diễn vector và mã định danh (\texttt{doc\_id}).
                    \item \textbf{Tính ổn định:} Giảm thiểu tần suất ghi (write operations); chỉ cập nhật khi nội dung cốt lõi thay đổi.
                    \item Giảm thiểu đáng kể chi phí tính toán trong quá trình Indexing.
                \end{itemize}
            \end{infobox}
        \end{column}%
    \end{columns}
\end{frame}

\begin{frame}{Phân hệ 2: Truy vấn \& Cơ chế Nới lỏng (Soft-Filtering)}
    \begin{infobox}[\faFilter \hspace{1pt} Logic Lọc Nới lỏng (Soft-Filtering)]
        \footnotesize Việc áp dụng các điều kiện cứng (Hard-filter) thường trả về tập rỗng do lỗi trích xuất thực thể. T-RAG áp dụng cơ chế tự động nới lỏng giới hạn nhằm duy trì một tập ứng viên (Candidate List) ổn định.
    \end{infobox}
    
    \vspace{0.1cm}
    
    \textbf{\small Minh họa quá trình Fallback:}
    \vspace{0.1cm}
    \begin{itemize}
        \small
        \item[\faFilter] Bước 1: Khớp toàn bộ điều kiện (\texttt{Date='2025'} AND \texttt{Author='A'} AND \texttt{Type='Meeting'}) 
        \item[] $\rightarrow$ Lượng kết quả thấp $\rightarrow$ Kích hoạt Fallback (Bỏ qua \texttt{Date}).
        \item[\faFilter] Bước 2: Khớp một phần (\texttt{Author='A'} AND \texttt{Type='Meeting'}) 
        \item[] $\rightarrow$ Lượng kết quả thấp $\rightarrow$ Kích hoạt Fallback (Bỏ qua \texttt{Author}).
        \item[\faFilter] Bước 3: Chỉ giữ thuộc tính cốt lõi (\texttt{Type='Meeting'})
        \item[] $\rightarrow$ \textcolor{mygreen}{\textbf{Đạt ngưỡng ứng viên (Ví dụ: 500 documents) $\rightarrow$ Chuyển tiếp tới Hybrid Retriever.}}
    \end{itemize}
\end{frame}

\begin{frame}{Phân hệ 3: Truy xuất Lai (Hybrid Retriever)}
    \begin{successbox}[\faSearch \hspace{2pt} Tích hợp Dense \& Sparse]
        \small Hoạt động truy xuất chỉ được thực hiện trên tập ứng viên giới hạn (Candidate List) thay vì toàn bộ cơ sở dữ liệu.
    \end{successbox}
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \small
        \item \textbf{Dense Search (Semantic):} Đánh giá độ tương đồng Cosine, cho phép ánh xạ các khái niệm tương đương dù không trùng khớp từ vựng.
        \item \textbf{Sparse Search (Lexical - BM25):} Cần thiết để nhận diện chính xác các mã định danh, ID đặc thù (như mã lỗi, số ticket) mà Dense Search thường bỏ sót.
        \item \textbf{Hợp nhất bằng RRF (Reciprocal Rank Fusion):} Kết hợp thứ hạng từ cả hai phương pháp để trích xuất tập tài liệu liên quan nhất.
    \end{itemize}
\end{frame}

\begin{frame}{Phân hệ 4: Mô hình Reranker (Temporal \& Diversity)}
    \begin{alertbox}[\faCalculator \hspace{2pt} Hàm phạt thời gian (Time Decay)]
        \vspace{-0.2cm}
        \begin{equation}
            \text{Final\_Score} = \text{Relevance} \times e^{-\lambda \Delta t} \times \text{Diversity\_Penalty}
        \end{equation}
    \end{alertbox}
    
    \vspace{0.2cm}
    
    \begin{itemize}
        \small
        \item \textbf{Relevance ($0 \to 1$):} Điểm đánh giá mức độ liên quan ngữ nghĩa từ mô hình Cross-Encoder (ví dụ: BGE-Reranker).
        \item \textbf{Time Penalty ($e^{-\lambda \Delta t}$):} Với $\Delta t$ là khoảng lùi thời gian. Hệ số này làm suy giảm theo hàm mũ điểm số của các tài liệu cũ, ưu tiên thứ hạng cho thông tin hiện hành.
        \item \textbf{Diversity Penalty (MMR):} Áp dụng thuật toán Maximal Marginal Relevance nhằm phạt các tài liệu có nội dung trùng lặp, đảm bảo tính đa dạng trong tập kết quả cuối.
    \end{itemize}
\end{frame}

\section{Thiết lập Môi trường \& Đo lường (Benchmark Setup)}

\begin{frame}{Mô hình Triển khai trên GPU H100 (1)}
    \begin{itemize}
        \item[\faMicrochip] \textbf{Phần cứng:} 1 Node GPU H100 (40GB - 80GB VRAM).
        \item[\faCode] \textbf{Nền tảng (Native Python + Asyncio):} Xây dựng luồng thực thi bất đồng bộ nhằm tối ưu hóa độ trễ (Latency) và tính minh bạch của quy trình.
        \item[\faTrophy] \textbf{Mục tiêu Đánh giá:} Tối đa hóa thông lượng (Throughput) trên tập dữ liệu EnterpriseRAG-Bench để phục vụ công bố kết quả thực nghiệm.
    \end{itemize}
\end{frame}

\begin{frame}{Mô hình Triển khai trên GPU H100 (2)}
    \begin{infobox}[\faCogs \hspace{2pt} Phân bổ Tài nguyên VRAM]
        \begin{itemize}
            \item \textbf{LLM Engine (vLLM):} Vận hành mô hình Llama-3-8B / Qwen2.5-7B, xử lý tác vụ \textit{Query Parsing} và \textit{Generation}. (Yêu cầu $\sim 16\text{GB}$). Sử dụng PagedAttention để tăng Batch Size.
            \item \textbf{Cross-Encoder:} Vận hành BGE-Reranker-v2 thông qua thư viện \texttt{Sentence-Transformers}. (Yêu cầu $\sim 4\text{GB}$).
            \item \textbf{Cơ sở Dữ liệu cục bộ:} 
            \begin{itemize}
                \item SQLite FTS5 (In-memory/SSD)
                \item LanceDB (In-memory/SSD)
                \item Các DB hoạt động bên ngoài VRAM nhằm tránh gây hiện tượng OOM (Out Of Memory).
            \end{itemize}
        \end{itemize}
    \end{infobox}
\end{frame}
