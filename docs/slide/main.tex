\begin{frame}{Stage 1: PSR Router \& Lý thuyết Adaptive Tau}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Công thức PSR Router (Logistic Regression):}
$$\boxed{P(s_i | Q) = \sigma(W_i \cdot \text{Encoder}(Q) + b_i)}$$

\textbf{Thuật toán Adaptive Tau (Dynamic Thresholding):}
\begin{enumerate}
    \item Shannon Entropy: $H = -\sum_{i=1}^K n_i \ln(n_i + 1e^{-10})$, với $n_i = p_i / \sum p_k$
    \item Confidence: $conf = 1.0 - \frac{H}{\ln(K)}$ \quad ($K=9$ shards)
    \item Ngưỡng hiệu dụng $\tau_{eff}$:
    $$\boxed{\tau_{eff} = \tau_{base} + \alpha \cdot (conf - 0.5)}$$
    kẹp an toàn trong khoảng $[0.05, 0.40]$.
\end{enumerate}
\end{column}

\begin{column}{0.44\textwidth}
\begin{exampleblock}{Tự điều chỉnh theo loại câu hỏi}
\begin{itemize}
    \item \textbf{Câu hỏi rõ nguồn}: Router dự đoán xác suất tập trung vào 1 shard $\rightarrow H$ thấp $\rightarrow conf$ cao $\rightarrow \tau_{eff}$ tăng lên $\rightarrow$ \textbf{Chỉ quét 1 Shard}.
    \item \textbf{Câu hỏi mơ hồ}: Xác suất san đều $\rightarrow H$ cao $\rightarrow conf$ thấp $\rightarrow \tau_{eff}$ hạ xuống $\rightarrow$ \textbf{Mở rộng quét 3-4 Shards} để bảo toàn Recall.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Ví dụ Tính toán (1/2): PSR Router \& Chuẩn hóa Phân phối}
\small
\textbf{Câu hỏi truy vấn ($Q$):} \textit{"Có ticket Jira nào liên quan đến lỗi out of memory của cổng OpenAI API được nhắc đến trên Slack không?"}

\vspace{0.2cm}
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{Bước 1: Dự đoán xác suất PSR $P(s_i|Q)$}
Sigmoid Logistic Regression dự đoán xác suất từng Shard chứa câu trả lời:
\begin{itemize}
    \item $P(\text{slack}|Q) = \mathbf{0.85}$ (Slack)
    \item $P(\text{jira}|Q) = \mathbf{0.75}$ (Jira)
    \item $P(\text{github}|Q) = 0.12$ (GitHub)
    \item $P(\text{confluence}|Q) = 0.05$ (Confluence)
    \item Các nguồn khác (\texttt{gmail, gdrive...}): $\le 0.02$
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{Bước 2: Phân phối chuẩn hóa $n_i = p_i / S$}
Tính tổng xác suất $S = \sum_{i=1}^9 p_i = \mathbf{1.83}$ và chuẩn hóa về phân phối đơn vị $n_i$:
\begin{itemize}
    \item $n_{\text{slack}} = 0.85 / 1.83 = \mathbf{0.464}$
    \item $n_{\text{jira}} = 0.75 / 1.83 = \mathbf{0.410}$
    \item $n_{\text{github}} = 0.12 / 1.83 = \mathbf{0.066}$
    \item $n_{\text{confluence}} = 0.05 / 1.83 = \mathbf{0.027}$
    \item $n_{\text{khác}} \le 0.011$ ($\sum_{i=1}^9 n_i = 1.0$)
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Ví dụ Tính toán (2/2): Adaptive Tau \& Chọn Active Shards}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{alertblock}{Bước 3: Tính toán Adaptive Tau}
\begin{enumerate}
    \item \textbf{Shannon Entropy}:
    $$H = -\sum_{i=1}^9 n_i \ln n_i \approx \mathbf{1.25}$$
    \item \textbf{Entropy tối đa}: $H_{max} = \ln(9) \approx \mathbf{2.197}$
    \item \textbf{Hệ số tự tin (Confidence)}:
    $$conf = 1.0 - \frac{1.25}{2.197} = \mathbf{0.431}$$
    \item \textbf{Ngưỡng hiệu dụng $\tau_{eff}$}:
    $$\tau_{eff} = 0.15 + 0.08 \times (0.431 - 0.5) = \mathbf{0.1445}$$
\end{enumerate}
\end{alertblock}
\end{column}

\begin{column}{0.46\textwidth}
\begin{block}{Ý nghĩa Tham số ($\tau_{base}=0.15, \alpha=0.08$)}
\begin{itemize}
    \item $\mathbf{\tau_{base} = 0.15}$: Ngưỡng định tuyến cơ sở được tối ưu qua Grid Search.
    \item $\mathbf{\alpha = 0.08}$: Hệ số điều chỉnh biên độ ($\pm 0.04$). Vì Router mơ hồ ($conf = 0.431 < 0.5$), $\tau_{eff}$ tự động hạ từ $0.15 \rightarrow 0.1445$ để mở rộng khung quét.
\end{itemize}
\end{block}

\begin{exampleblock}{Kết quả Chọn Active Shards ($P \ge 0.1445$)}
$\rightarrow$ Active Shards: \textbf{\{\texttt{slack}, \texttt{jira}\}} (2/9 shards).\\
Đóng băng 7/9 shards, tiết kiệm \textbf{75\% Search Space}!
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Stage 2 \& 3: Local Hybrid Search, SW-RRF \& Smart Hop 2}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{SW-RRF Fusion \& Hybrid Search}
$$\boxed{Score(d) = P(s_d|Q)^\gamma \cdot \left( \frac{W_{dense}}{k + r_{dense}} + \frac{W_{sparse}}{k + r_{sparse}} \right)}$$
\begin{itemize}
    \item \textbf{Dense Search}: LanceDB IVF-PQ index.
    \item \textbf{Sparse Search}: BM25 Tantivy FTS.
    \item Trọng số tối ưu: $W_{dense}=0.3, W_{sparse}=0.7$.
    \item Bayesian Prior $\gamma=0.4 \div 0.5$: Ưu tiên shard có xác suất Router cao.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Smart Hop 2 Decision (2 Điều kiện SKIP)}
\begin{enumerate}
    \item \textbf{Đơn nguồn}: $active\_shards\_count < 2 \rightarrow$ \textbf{SKIP Hop 2}.
    \item \textbf{Match Hop 1 xuất sắc}: $top1\_dist < 0.55 \rightarrow$ \textbf{SKIP Hop 2}.
\end{enumerate}
\end{alertblock}

\begin{exampleblock}{Ví dụ tiếp tục (Q ví dụ trên):}
\begin{itemize}
    \item $active\_shards = 2 \ge 2$ (Chưa SKIP1).
    \item $top1\_dist = 0.58 \ge 0.55$ (Chưa SKIP2).
    \item $\rightarrow$ \textbf{TRIGGER Hop 2}: Trích xuất thực thể \texttt{"SDK-402, OpenAI API"} qua LLM $\rightarrow$ Tìm thêm được PR Link liên quan trên Jira.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}
