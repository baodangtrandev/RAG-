\section{Tính Mới \& Đóng góp Khoa học của T-RAG v2}

\begin{frame}{4 Cải tiến Thế hệ Mới trong T-RAG v2}
\small
\begin{columns}[T]
\begin{column}{0.49\textwidth}
\begin{block}{\#1 --- PSR Router + Adaptive Tau}
Tự động điều chỉnh ngưỡng $\tau_{eff} = \tau_{base} + \alpha \cdot (conf - 0.5)$ theo độ tự tin của Router. Giảm không gian tìm kiếm từ \textbf{36\% đến 67\%}.
\end{block}

\vspace{0.1cm}
\begin{block}{\#2 --- Sửa lỗi Double-Encode (Tăng tốc 6x)}
Tái sử dụng vector embedding giữa Hop 1 và Hop 2, kéo Retrieval Latency từ \textbf{1.68s} xuống còn \textbf{0.27s}.
\end{block}
\end{column}

\begin{column}{0.49\textwidth}
\begin{block}{\#3 --- Cơ chế Smart Hop 2 (Conditional)}
Bỏ qua Hop 2 nếu câu hỏi đơn nguồn ($active\_shards < 2$) hoặc Hop 1 đã rất khớp ($top1\_dist < 0.55$), tiết kiệm \textbf{34\% Total Latency}.
\end{block}

\vspace{0.1cm}
\begin{block}{\#4 --- SW-RRF + Hybrid Search Tuning}
Tăng trọng số BM25 trên từ khóa kỹ thuật ($D=0.3, S=0.7$) kết hợp với Bayesian Prior $P(\text{Source}|Q)^\gamma$.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Stage 1: PSR Router \& Ngưỡng động Adaptive Tau}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Công thức PSR Router:}
$$\boxed{P(s_i | Q) = \sigma(W_i \cdot \text{Encoder}(Q) + b_i)}$$

\textbf{Thuật toán Adaptive Tau (Dynamic Thresholding):}
\begin{enumerate}
    \item Shannon Entropy: $H = -\sum_{i=1}^K p_i \ln(p_i)$
    \item Confidence: $conf = 1.0 - \frac{H}{\ln(K)}$
    \item Ngưỡng hiệu dụng $\tau_{eff}$:
    $$\boxed{\tau_{eff} = \tau_{base} + \alpha \cdot (conf - 0.5)}$$
    $\tau_{eff}$ được kẹp trong khoảng $[0.05, 0.40]$.
\end{enumerate}
\end{column}
\begin{column}{0.44\textwidth}
\begin{exampleblock}{Cơ chế tự thích ứng theo câu hỏi}
\begin{itemize}
    \item \textbf{Câu hỏi rõ ràng} (\textit{"Fix GitHub Actions pipeline error"}):\\
    Entropy $H$ thấp $\rightarrow conf = 0.84 \rightarrow \tau_{eff} = 0.177 \rightarrow$ Chỉ quét 1 bảng \texttt{github}.
    \item \textbf{Câu hỏi mơ hồ} (\textit{"API discussion yesterday"}):\\
    Entropy $H$ cao $\rightarrow conf = 0.11 \rightarrow \tau_{eff} = 0.119 \rightarrow$ Quét rộng 4 bảng để tăng Recall.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Stage 2 \& 3: Local Hybrid Search, SW-RRF \& Smart Hop 2}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{SW-RRF Fusion \& Hybrid Search}
$$\boxed{Score(d) = P(s_d|Q)^\gamma \cdot \left( \frac{W_{dense}}{k + r_{dense}} + \frac{W_{sparse}}{k + r_{sparse}} \right)}$$
\begin{itemize}
    \item \textbf{Dense Search}: LanceDB IVF-PQ index.
    \item \textbf{Sparse Search}: BM25 Tantivy FTS.
    \item Trọng số tối ưu: $W_{dense}=0.3, W_{sparse}=0.7$ ($D=0.3, S=0.7$).
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Smart Hop 2 Decision (2 Điều kiện SKIP)}
\begin{enumerate}
    \item \textbf{Đơn nguồn}: $active\_shards\_count < 2 \rightarrow$ \textbf{SKIP Hop 2}.
    \item \textbf{Match Hop 1 xuất sắc}: $top1\_dist < 0.55 \rightarrow$ \textbf{SKIP Hop 2}.
\end{enumerate}
Nếu không thỏa mãn $\rightarrow$ Call LLM batch entity extraction $\rightarrow$ Hop 2 Search $\rightarrow$ Merge \& Dedup.
\end{alertblock}
\end{column}
\end{columns}
\end{frame}
