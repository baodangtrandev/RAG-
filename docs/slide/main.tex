\begin{columns}

\begin{block}{Dataset: EnterpriseRAG-Bench}
\begin{itemize}
    \item \textbf{511,962 documents} across 9 different sources
    \item Slack (285k), Gmail (121k), Jira (41k),\\Confluence (5k), GitHub, Fireflies...
    \item \textbf{500 benchmark queries}, 10 categories
    \item Noisy Data: Semantic conflicts, duplicated information
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.5\textwidth}
\begin{alertblock}{3 Major Limitations of Standard RAG}
\begin{enumerate}
    \item \textbf{Source Ambiguity}: Scatters embedding space representation and increases retrieval noise.
    \item \textbf{Vocabulary Mismatch}: Gap between natural queries and technical terms.
    \item \textbf{Multi-source Queries}: Requires complex cross-source reasoning.
\end{enumerate}
\end{alertblock}
\end{column}
\end{columns}

\vspace{0.4cm}
\begin{center}
\textit{Standard RAG architectures suffer from high search complexity and noise when searching the global document space.}
\end{center}
\end{frame}

\section{Novelty \& Contributions}

\begin{frame}{3 Academic Contributions}

\begin{block}{\#1 --- Probabilistic Source Router (PSR) + Database Sharding}
\textbf{System-ML Co-design}: Co-designing vector database sharding with routing models to reduce the document search space by \textbf{50--90\%}.
\end{block}

\vspace{0.2cm}
\begin{block}{\#2 --- Source-Weighted RRF (SW-RRF)}
\textbf{Mathematical Formulation}: Integrates a \textbf{Bayesian Prior} directly into Reciprocal Rank Fusion to optimize multi-source candidate ranking.
\end{block}

\vspace{0.2cm}
\begin{block}{\#3 --- Cross-Source Entity Propagation (CSEP)}
\textbf{Multi-step Reasoning}: Solves cross-source dependencies by extracting anchor entities to guide secondary reasoning hops.
\end{block}
\end{frame}

\begin{frame}{Stage 1: Probabilistic Source Router (PSR)}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Routing Formulation:} \\
Predicting source conditional probabilities:
\vspace{-0.2cm}
{\footnotesize $$\boxed{P(s_i | Q) = \text{Softmax}(W \cdot \text{Encoder}(Q) + b)_i}$$}

\textbf{Sub-space Filtering (Sharding):}
\begin{itemize}
    \item Activating queries only on selected sub-spaces:
\vspace{-0.2cm}
    {\footnotesize $$S_{active} = \{ s_i \in S \mid P(s_i | Q) \ge \tau \}$$}
\end{itemize}

\vspace{0.1cm}
\textbf{Key Outcome:} \\
Search complexity reduced from $O(|D_{\text{total}}|)$ to $O(\sum_{i \in S_{\text{active}}} |D_{s_i}|)$, drastically reducing I/O latency.
\end{column}
\begin{column}{0.44\textwidth}
\begin{exampleblock}{Routing Illustration (Example)}
Query: \textit{"What did the team say about the GPU budget meeting?"}

\vspace{0.1cm}
PSR conditional probability distribution:
\begin{itemize}
    \item $P(\text{Fireflies}|Q) = 0.81 > \tau \rightarrow$ Active
    \item $P(\text{Slack}|Q) = 0.72 > \tau \rightarrow$ Active
    \item $P(\text{Jira}|Q) = 0.04 \le \tau \rightarrow$ Pruned
    \item $P(\text{GitHub}|Q) = 0.02 \le \tau \rightarrow$ Pruned
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Stage 2: Source-Weighted RRF (SW-RRF)}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Standard RRF (Cormack et al., 2009):}\\
Scores candidates uniformly, failing to prioritize primary source context.

\vspace{0.2cm}
\textbf{Proposed Innovation (SW-RRF):}\\
Incorporates the routing probability $P(s_d|Q)$ as a Bayesian Prior.

\vspace{-0.2cm}
{\footnotesize $$\boxed{Score(d) = P(s_d|Q)^\gamma \cdot \left( \frac{\alpha}{k + r_{dense}} + \frac{1-\alpha}{k + r_{sparse}} \right)}$$}

\begin{itemize}
    \item $\gamma$: Source Bias Factor
    \item $\alpha$: Dense/Sparse weights
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Theoretical Foundation of SW-RRF}
Optimizes candidate ranking by adjusting score distributions based on source likelihood ($P(s_d|Q)$), eliminating surface-level keyword distractors from irrelevant sources.
\end{block}

\vspace{0.1cm}
\begin{exampleblock}{Hybrid Search Synergy}
\begin{itemize}
    \item \textbf{Dense Search}: Captures abstract semantic representation.
    \item \textbf{Sparse (BM25/FTS)}: Restricts search with exact keyword matching.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Stage 3 \& 4: Multi-hop \& Reranking}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{Stage 3: CSEP (Multi-hop Reasoning)}
Approximated via Bipartite Graph Random Walk.
\begin{enumerate}
    \item \textbf{Hop 1:} Retrieve anchor documents ($D_{\text{anchor}}$) from the highest-probability sources.
    \item \textbf{Entity Extraction}: Extract and identify unique anchor entities $E$ from $D_{\text{anchor}}$.
    \item \textbf{Hop 2:} Query expansion $Q^{(2)} = Q \oplus E$ to bridge related contexts cross-source.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Stage 4: Confidence Thresholding}
Leverages a \textbf{Cross-Encoder} to compute accurate relevance score $CE(Q, d_{\text{final}})$.

\vspace{0.1cm}
\textbf{Hard Thresholding}: Prunes low-confidence documents if $CE(Q, d) < \theta$.

\vspace{0.1cm}
If $D_{\text{final}} = \emptyset$, early-exit directly returns "Unanswerable", successfully minimizing hallucinations.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Experimental Results Analysis}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{block}{Computational Efficiency (Speedup)}
\begin{itemize}
    \item Standard Vector Reranker takes \textbf{12.50s} due to full-database scan.
    \item T-RAG [High-Speed] takes only \textbf{0.77s} (\textbf{16.2x speedup}) by querying targeted shards and bypassing hop 2.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.5\textwidth}
\begin{block}{Relevance \& Accuracy (Correctness)}
\begin{itemize}
    \item T-RAG [High-Recall] reaches \textbf{32.0\%} correctness (outperforming Vector Reranker's 28.5\% and BM25's 27.8\%).
    \item Source penalty ($\gamma=1.0$) filters noisy text, boosting correctness from \textbf{29.6\% to 31.1\%}.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Conclusion \& Future Work}
\begin{block}{Conclusion}
\begin{itemize}
    \item T-RAG V2 (Hybrid Search \& SW-RRF) effectively resolves multi-source enterprise RAG challenges.
    \item Delivers outstanding performance (\textbf{>16x speedup}) while surpassing standard Vector Reranker accuracy.
\end{itemize}
\end{block}

\vspace{0.2cm}
\begin{block}{Future Directions (V3)}
\begin{itemize}
    \item Fine-tuning RRF weights (Dense 0.3 / Sparse 0.7) to improve exact keyword recall.
    \item Tracking the physical search space (in docs) and actual refusal rates to better control hallucinations.
\end{itemize}
\end{block}
\end{frame}
