\begin{column}{0.5\textwidth}
\begin{alertblock}{3 Hạn chế cốt lõi của Standard RAG}
\begin{enumerate}
    \item \textbf{Mơ hồ nguồn (Source Ambiguity)}: Làm loãng không gian vector embedding và tăng nhiễu khi truy vấn.
    \item \textbf{Bất đồng nhất từ vựng (Vocabulary Mismatch)}: Khoảng cách lớn giữa câu hỏi tự nhiên và thuật ngữ kỹ thuật trong tài liệu.
    \item \textbf{Truy vấn đa nguồn (Multi-source Queries)}: Yêu cầu khả năng lập luận, liên kết chéo thông tin từ nhiều nguồn.
\end{enumerate}
\end{alertblock}
\end{column}
\end{columns}

\vspace{0.4cm}
\begin{center}
\textit{Kiến trúc Standard RAG gặp hạn chế lớn về độ phức tạp tính toán và nhiễu thông tin khi phải tìm kiếm trên toàn bộ không gian dữ liệu toàn cục.}

\section{Tính Mới \& Đóng góp Khoa học}

\begin{frame}{3 Đóng góp Khoa học Cốt lõi}

\begin{block}{\#1 --- Bộ định tuyến nguồn xác suất (PSR) + Database Sharding}
\textbf{System-ML Co-design}: Tích hợp Database Sharding với PSR để giảm không gian tìm kiếm từ \textbf{50--90\%}.
\end{block}

\vspace{0.1cm}
\begin{block}{\#2 --- Thuật toán Source-Weighted RRF (SW-RRF)}
\textbf{Mô hình Toán học}: Tích hợp tiền nghiệm Bayesian vào RRF giúp tối ưu hóa thứ hạng ứng viên đa nguồn.
\end{block}

\vspace{0.1cm}
\begin{block}{\#3 --- Cơ chế lan truyền thực thể đa nguồn (CSEP)}
\textbf{Lập luận Đa bước}: Giải quyết phụ thuộc chéo bằng cách trích xuất Anchor Entities để dẫn hướng cho truy vấn thứ cấp (Hop 2).
\end{block}
\end{frame}

\begin{frame}{Stage 1: Bộ định tuyến nguồn xác suất (PSR)}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Công thức định tuyến:} \\
Dự đoán xác suất có điều kiện của nguồn dữ liệu:
\vspace{-0.2cm}
{\footnotesize $$\boxed{P(s_i | Q) = \text{Softmax}(W \cdot \text{Encoder}(Q) + b)_i}$$}

\textbf{Lọc không gian con (Database Sharding):}
\begin{itemize}
    \item Chỉ kích hoạt truy vấn trên các phân mảnh được lựa chọn:
\vspace{-0.2cm}
    {\footnotesize $$S_{active} = \{ s_i \in S \mid P(s_i | Q) \ge \tau \}$$}
\end{itemize}

\vspace{0.1cm}
\textbf{Kết quả chính:} \\
Độ phức tạp tìm kiếm giảm từ toàn cục xuống cục bộ (theo phân mảnh active), tiết kiệm tối đa tài nguyên I/O.
\end{column}
\begin{column}{0.44\textwidth}
\begin{exampleblock}{Minh họa cơ chế Định tuyến}
Query: \textit{"What did the team say about the GPU budget meeting?"}

\vspace{0.1cm}
Phân phối xác suất có điều kiện từ PSR:
\begin{itemize}
    \item $P(\text{Fireflies}|Q) = 0.81 > \tau \rightarrow$ Kích hoạt (Active)
    \item $P(\text{Slack}|Q) = 0.72 > \tau \rightarrow$ Kích hoạt (Active)
    \item $P(\text{Jira}|Q) = 0.04 \le \tau \rightarrow$ Loại bỏ (Pruned)
    \item $P(\text{GitHub}|Q) = 0.02 \le \tau \rightarrow$ Loại bỏ (Pruned)
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Stage 2: Thuật toán Source-Weighted RRF (SW-RRF)}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Thuật toán RRF chuẩn (Cormack và cộng sự, 2009):}\\
Đánh giá các ứng viên đồng đều, chưa phân biệt tầm quan trọng của nguồn dữ liệu mục tiêu.

\vspace{0.2cm}
\textbf{Cải tiến đề xuất (SW-RRF):}\\
Tích hợp xác suất định tuyến $P(s_d|Q)$ làm trọng số ưu tiên Bayesian Prior.

\vspace{-0.2cm}
{\footnotesize $$\boxed{Score(d) = P(s_d|Q)^\gamma \cdot \left( \frac{\alpha}{k + r_{dense}} + \frac{1-\alpha}{k + r_{sparse}} \right)}$$}

\begin{itemize}
    \item $\gamma$: Hệ số khuếch đại nguồn (Source Bias Factor)
    \item $\alpha$: Trọng số Dense/Sparse (ví dụ: 0.3 / 0.7)
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Cơ sở lý thuyết của SW-RRF}
Tối ưu hóa thứ hạng bằng cách tinh chỉnh điểm số theo độ tin cậy của nguồn, giảm thiểu nhiễu từ khóa bề mặt từ các nguồn không liên quan.
\end{block}

\vspace{0.1cm}
\begin{exampleblock}{Sự kết hợp trong Hybrid Search}
\begin{itemize}
    \item \textbf{Dense Search}: Ghi nhận biểu diễn ngữ nghĩa trừu tượng.
    \item \textbf{Sparse Search (BM25/FTS)}: Thu hẹp phạm vi tìm kiếm bằng khớp từ khóa chính xác.
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Stage 3 \& 4: Lập luận đa bước \& Tái xếp hạng}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{Stage 3: Thuật toán CSEP (Lập luận đa bước)}
Mô phỏng qua mô hình duyệt ngẫu nhiên trên đồ thị hai phía (Bipartite Graph Random Walk).
\begin{enumerate}
    \item \textbf{Hop 1:} Truy vấn các tài liệu neo ($D_{\text{anchor}}$) từ các nguồn có xác suất cao nhất.
    \item \textbf{Entity Extraction}: Trích xuất và nhận diện các thực thể neo $E$ độc bản từ $D_{\text{anchor}}$ bằng LLM.
    \item \textbf{Hop 2:} Mở rộng truy vấn $Q^{(2)} = Q \oplus E$ để kết nối các ngữ cảnh liên quan chéo nguồn.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Stage 4: Xác thực và Đánh giá Độ tin cậy}
Sử dụng mô hình \textbf{Cross-Encoder} để tính toán điểm tương đồng ngữ nghĩa chính xác $CE(Q, d_{\text{final}})$.

\vspace{0.1cm}
\textbf{Ngưỡng lọc cứng (Hard Thresholding)}: Lọc bỏ các tài liệu có độ tin cậy thấp nếu điểm số $CE(Q, d) < \theta$.

\vspace{0.1cm}
Nếu tập tài liệu cuối cùng $D_{\text{final}} = \emptyset$, cơ chế thoát sớm (Early-Exit) sẽ trả về trực tiếp thông báo từ chối trả lời (Unanswerable), giúp loại bỏ ảo tưởng thông tin (Hallucination).
\end{block}
\end{column}
\end{columns}
\end{frame}
