\section{Tính Mới \& Đóng góp (Academic Novelty)}

\begin{frame}{3 Đóng góp Học thuật (Novelty)}

\begin{block}{\#1 --- Probabilistic Source Router (PSR) + Database Sharding}
\textbf{System-ML Co-design}: Phân vùng vật lý Vector DB theo nguồn. PSR dự đoán xác suất định tuyến, giúp giảm không gian tìm kiếm (Search Space) từ \textbf{50--90\%}.
\end{block}

\vspace{0.2cm}
\begin{block}{\#2 --- Source-Weighted RRF (SW-RRF)}
\textbf{Đề xuất toán học mới}: Tích hợp \textbf{Bayesian Prior} (xác suất nguồn) trực tiếp vào công thức Reciprocal Rank Fusion truyền thống để tối ưu lai ghép đa nguồn.
\end{block}

\vspace{0.2cm}
\begin{block}{\#3 --- Cross-Source Entity Propagation (CSEP)}
\textbf{Thuật toán Multi-hop mới}: Giải quyết bài toán truy vấn chéo nguồn bằng cách lấy thông tin mỏ neo (Anchor) từ một nguồn để mở rộng tìm kiếm sang nguồn khác.
\end{block}
\end{frame}

\begin{frame}{Giai đoạn 1: Probabilistic Source Router (PSR)}
\small
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\textbf{Toán học định tuyến:} 
Dự báo xác suất có điều kiện của nguồn:
\vspace{-0.2cm}
{\footnotesize $$\boxed{P(s_i | Q) = \text{Softmax}(W \cdot \text{Encoder}(Q) + b)_i}$$}

\textbf{Thuật toán Phân vùng (Sharding):}
\begin{itemize}
    \item Chỉ kích hoạt tìm kiếm trên tập nguồn con:
\vspace{-0.2cm}
    {\footnotesize $$S_{active} = \{ s_i \in S \mid P(s_i | Q) \ge \tau \}$$}
\end{itemize}

\vspace{0.1cm}
\textbf{Kết quả:} 
Độ phức tạp giảm từ $O(|D_{total}|)$ xuống $O(\sum_{i \in S_{active}} |D_{s_i}|)$. Giảm I/O đáng kể.
\end{column}
\begin{column}{0.44\textwidth}
\begin{exampleblock}{Ví dụ}
Truy vấn: \textit{"What did the team say about the GPU budget meeting?"}

\vspace{0.1cm}
PSR output:
\begin{itemize}
    \item $P(\text{Fireflies}|Q) = 0.81 \rightarrow$ Chọn
    \item $P(\text{Slack}|Q) = 0.72 \rightarrow$ Chọn
    \item $P(\text{Jira}|Q) = 0.04 \rightarrow$ Bỏ
    \item $P(\text{GitHub}|Q) = 0.02 \rightarrow$ Bỏ
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Giai đoạn 2: Source-Weighted RRF (SW-RRF)}
\small
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{RRF truyền thống (Cormack, 2009):}
Đánh đồng mọi tài liệu, làm mất đi sự "ưu tiên" cho nguồn chính.

\vspace{0.2cm}
\textbf{Đề xuất tính mới (SW-RRF):}
Đưa xác suất $P(s_d|Q)$ làm \textbf{Bayesian Prior} vào công thức xếp hạng.

\vspace{-0.2cm}
{\footnotesize $$\boxed{Score(d) = P(s_d|Q)^\gamma \cdot \left( \frac{\alpha}{k + r_{dense}} + \frac{1-\alpha}{k + r_{sparse}} \right)}$$}

\begin{itemize}
    \item $\gamma$: Source Bias Factor
    \item $\alpha$: Trọng số Dense/Sparse
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Tại sao SW-RRF hiệu quả?}
Tài liệu thuộc nguồn có xác suất cao (từ PSR) sẽ được \textbf{"boost"} điểm toán học. Ép các tài liệu đúng nguồn trồi lên top, đánh bại các tài liệu đồng nghĩa nhưng nằm sai nguồn.
\end{block}

\vspace{0.1cm}
\begin{exampleblock}{Sự cần thiết của Hybrid}
\begin{itemize}
    \item \textbf{Dense Search}: Khớp ngữ nghĩa
    \item \textbf{Sparse (BM25)}: Bắt chính xác (Exact match)
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}
\end{frame}

%------------------------------------------------
\begin{frame}{Giai đoạn 3 \& 4: Multi-hop \& Reranking}
\small
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\begin{block}{Giai đoạn 3: CSEP (Multi-hop)}
Dành cho truy vấn đa nguồn phức tạp.
\begin{enumerate}
    \item \textbf{Hop 1:} Tìm trên nguồn có $P(s|Q)$ cao nhất $\rightarrow$ Lấy tài liệu mỏ neo ($D_{anchor}$).
    \item \textbf{Entity Extract:} Trích xuất thực thể $E$ từ $D_{anchor}$.
    \item \textbf{Hop 2:} Cập nhật truy vấn $Q^{(2)} = Q \oplus E$, tìm trên nguồn thứ 2.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\begin{block}{Giai đoạn 4: Đánh giá độ tin cậy}
Dùng \textbf{Cross-Encoder} tính toán hàm liên kết $CE(Q, d_{final})$.

\vspace{0.1cm}
Áp dụng \textbf{Hard Thresholding}: Lọc bỏ tài liệu nếu $CE(Q, d) < \theta$.

\vspace{0.1cm}
Nếu rỗng $\rightarrow$ Trả về "Unanswerable" $\rightarrow$ \textbf{Giảm Hallucination triệt để!}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Kế hoạch Thực thi Kỹ thuật}

\begin{frame}{Kế hoạch Thực thi \& Đánh giá}

\begin{block}{Các bước Triển khai}
\begin{enumerate}
    \item \textbf{Chuẩn bị Dữ liệu}: Sharding 511k documents thành các bảng độc lập trong \textbf{LanceDB} theo \texttt{source\_type}.
    \item \textbf{Train PSR Router}: Fine-tune LLM nhỏ (Llama-3-8B hoặc BERT) để làm bộ Classifier.
    \item \textbf{Cài đặt SW-RRF}: Lập trình hàm Retriever theo đúng phương trình toán học đã đề xuất.
\end{enumerate}
\end{block}

\vspace{0.3cm}
\begin{block}{Đánh giá (So với Standard RAG)}
\begin{itemize}
    \item \textbf{Chất lượng:} Recall@K, NDCG@10 (sử dụng tập ground truth)
    \item \textbf{Tốc độ/Chi phí:} Latency (ms), Search Space Size (\% documents được load)
\end{itemize}
\end{block}
